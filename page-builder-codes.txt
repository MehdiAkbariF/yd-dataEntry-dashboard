import { Code2, WrapText, Copy, Check, Maximize2, Minimize2 } from 'lucide-react';

loader.config({ monaco });

interface CodeMirrorEditorProps {
  value: string;
  onChange: (value: string) => void;
  language?: 'html' | 'css';
  label?: string;
}

export function CodeMirrorEditor({
  value,
  onChange,
  language = 'html',
  label,
}: CodeMirrorEditorProps) {
  const [copied, setCopied] = useState(false);
  const [wrap, setWrap] = useState(true);
  const [fullscreen, setFullscreen] = useState(false);

  const handleCopy = async () => {
    try {
      await navigator.clipboard.writeText(value);
      setCopied(true);
      setTimeout(() => setCopied(false), 2000);
    } catch {}
  };

  return (
    <div
      className={
        fullscreen
          ? 'fixed inset-4 z-[999] shadow-2xl flex flex-col bg-neutral-950 border border-neutral-800 rounded-xl overflow-hidden'
          : 'flex flex-col h-full w-full bg-neutral-950 border border-neutral-800 rounded-xl overflow-hidden'
      }
    >
      {/* Header (RTL مناسب فارسی) */}
      <div 
        dir="rtl" 
        className="flex items-center justify-between px-3 py-2 border-b border-neutral-800 bg-neutral-900 shrink-0"
      >
        <div className="flex items-center gap-2">
          <Code2 className="h-3.5 w-3.5 text-amber-500" />
          <span className="text-[11px] font-bold text-white">{label}</span>
          <span className="text-[9px] font-mono text-neutral-500 uppercase px-1.5 py-0.5 rounded bg-neutral-800">
            {language}
          </span>
        </div>
        <div className="flex items-center gap-1">
          <button
            type="button"
            onClick={() => setWrap((w) => !w)}
            className={`p-1.5 rounded-md ${
              wrap ? 'text-amber-400 bg-amber-500/10' : 'text-neutral-400 hover:bg-neutral-800 hover:text-white'
            }`}
            title="Word Wrap"
          >
            <WrapText className="h-3.5 w-3.5" />
          </button>
          <button
            type="button"
            onClick={handleCopy}
            className="p-1.5 rounded-md text-neutral-400 hover:bg-neutral-800 hover:text-white"
            title="کپی"
          >
            {copied ? <Check className="h-3.5 w-3.5 text-emerald-400" /> : <Copy className="h-3.5 w-3.5" />}
          </button>
          <button
            type="button"
            onClick={() => setFullscreen((f) => !f)}
            className="p-1.5 rounded-md text-neutral-400 hover:bg-neutral-800 hover:text-white"

import { useMutation, useQueryClient } from '@tanstack/react-query';
import { pageService } from '../services/pageService';
import { CreatePageRequest, UpdatePageRequest } from '../types/schema.types';
import { pageBuilderKeys } from './usePages';

export function useCreatePage() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: ({
      request,
      categoryId,
    }: {
      request: CreatePageRequest;
      categoryId: string;
    }) => pageService.create(request, categoryId),
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: pageBuilderKeys.lists() });
    },
  });
}

export function useUpdatePage() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: (request: UpdatePageRequest) => pageService.update(request),
    onSuccess: (data) => {
      if (data) {
        qc.invalidateQueries({
          queryKey: pageBuilderKeys.detail(data.meta.id),
        });
      }
      qc.invalidateQueries({ queryKey: pageBuilderKeys.lists() });
    },
  });
}

export function useDeletePage() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: (id: string) => pageService.remove(id),
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: pageBuilderKeys.lists() });
    },
  });
}

import { useQuery } from '@tanstack/react-query';
import { pageService } from '../services/pageService';
import { FullPage, PageListItem } from '../types/schema.types';

export const pageBuilderKeys = {
  all: ['page-builder'] as const,
  lists: () => [...pageBuilderKeys.all, 'list'] as const,
  list: (params?: Record<string, unknown>) =>
    [...pageBuilderKeys.lists(), params] as const,
  details: () => [...pageBuilderKeys.all, 'detail'] as const,
  detail: (id: string) => [...pageBuilderKeys.details(), id] as const,
  categories: () => [...pageBuilderKeys.all, 'categories'] as const,
};

export function usePagesList(params?: {
  title?: string;
  pageNumber?: number;
  pageSize?: number;
}) {
  return useQuery<PageListItem[]>({
    queryKey: pageBuilderKeys.list(params),
    queryFn: () => pageService.getList(params),
    staleTime: 30 * 1000,
  });
}

export function usePageDetails(id: string) {
  return useQuery<FullPage | null>({
    queryKey: pageBuilderKeys.detail(id),
    queryFn: () => pageService.getById(id),
    enabled: !!id,
    staleTime: 60 * 1000,
  });
}

export function usePageCategories() {
  return useQuery({
    queryKey: pageBuilderKeys.categories(),
    queryFn: () => pageService.getCategories(),
    staleTime: 10 * 60 * 1000,
  });
}

interface BuildDocOptions {
  html: string;
  css: string;
  includeTailwind?: boolean;
  includeFonts?: boolean;
}

export function buildPreviewDoc({
  html,
  css,
  includeTailwind = true,
  includeFonts = true,
}: BuildDocOptions): string {
  const tailwind = includeTailwind
    ? '<script src="https://cdn.tailwindcss.com"></script>'
    : '';

  const fonts = includeFonts
    ? '<link href="https://cdn.jsdelivr.net/gh/rastikerdar/vazirmatn@v33.003/Vazirmatn-font-face.css" rel="stylesheet" />'
    : '';

  return `<!DOCTYPE html>
<html lang="fa" dir="rtl">
<head>
  <meta charset="UTF-8" />
  <meta name="viewport" content="width=device-width, initial-scale=1.0" />
  ${fonts}
  ${tailwind}
  <style>
    *, *::before, *::after { box-sizing: border-box; }
    html, body {
      margin: 0;
      padding: 0;
      min-height: 100%;
      font-family: 'Vazirmatn', system-ui, -apple-system, sans-serif;
      line-height: 1.6;
      color: #111;
      background-color: #ffffff;
    }
    img { max-width: 100%; height: auto; }
    /* استایل‌های سفارشی شما */
    ${css || ''}
  </style>
</head>
<body>
  ${html || ''}
</body>
</html>`;
}

// src/features/page-builder/lib/legacyMigration.ts
import type { PageContent } from '../types/schema.types';

/**
 * تشخیص اینکه محتوا Puck قدیمی است یا خیر
 */
export function isLegacyPuckContent(raw: string): boolean {
  return (
    typeof raw === 'string' &&
    (raw.startsWith('__PUCK__:') || raw.trim().startsWith('{"root":'))
  );
}

/**
 * تبدیل Puck JSON قدیمی به HTML/CSS معادل
 */
export function migratePuckToHtml(raw: string): PageContent {
  try {
    const jsonStr = raw.startsWith('__PUCK__:')
      ? raw.slice('__PUCK__:'.length)
      : raw;
    const data = JSON.parse(jsonStr);

    const rootProps = data?.root?.props || {};
    const items = Array.isArray(data?.content) ? data.content : [];

    const htmlParts = items
      .map((node: any) => renderNode(node))
      .filter(Boolean);

    // استخراج تنظیمات root
    const maxWidth = rootProps?.maxWidth || 'lg';
    const bg = rootProps?.backgroundColor || '';

    const MAX_WIDTH_MAP: Record<string, string> = {
      sm: 'max-w-2xl',
      md: 'max-w-4xl',
      lg: 'max-w-6xl',
      xl: 'max-w-7xl',
      full: 'max-w-full',
    };

    const containerClass = MAX_WIDTH_MAP[maxWidth] || 'max-w-6xl';

    const html = `<div${bg ? ` style="background-color:${bg};"` : ''}>
  <div class="page-container ${containerClass}">
${htmlParts.map((p) => `    ${p}`).join('\n')}
  </div>
</div>`;

    const css = `.page-container {
  margin: 0 auto;
  padding: 2rem 1rem;
}`;

    return { html, css };
  } catch (e) {
    console.error('[legacyMigration] failed:', e);
    return { html: '', css: '' };
  }
}

// ─────────────────────────────────────────
// Renderer هر node
// ─────────────────────────────────────────
function renderNode(node: any): string {
  if (!node || !node.type) return '';
  const props = node.props || {};

  switch (node.type) {
    case 'Heading':
      return renderHeading(props);
    case 'Text':
      return renderText(props);
    case 'Button':
      return renderButton(props);
    case 'Image':
      return renderImage(props);
    case 'Divider':
      return renderDivider(props);
    case 'Spacer':
      return renderSpacer(props);
    case 'Section':
      return renderSection(props);
    case 'RawHtml':
      return props.html || '';
    default:
      return `<!-- unsupported block: ${node.type} -->`;
  }
}

function alignClass(align: string): string {
  switch (align) {
    case 'center':
      return 'text-center';
    case 'left':
      return 'text-left';
    case 'right':
    default:
      return 'text-right';
  }
}

function escapeHtml(s: string): string {
  return String(s)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;');
}

function renderHeading(props: any): string {
  const level = props.level || 'h2';
  const text = escapeHtml(props.text || '');
  const color = props.color ? ` style="color:${props.color};"` : '';
  const align = alignClass(props.align);
  return `<${level} class="${align}"${color}>${text}</${level}>`;
}

function renderText(props: any): string {
  const content = escapeHtml(props.content || '');
  const align = alignClass(props.align);
  const color = props.color ? ` style="color:${props.color};"` : '';
  return `<p class="${align}"${color}>${content}</p>`;
}

function renderButton(props: any): string {
  const label = escapeHtml(props.label || 'دکمه');
  const href = escapeHtml(props.href || '#');
  const align = alignClass(props.align);
  return `<div class="${align}"><a href="${href}" class="btn">${label}</a></div>`;
}

function renderImage(props: any): string {
  if (!props.src) return '';
  const src = escapeHtml(props.src);
  const alt = escapeHtml(props.alt || '');
  const caption = props.caption
    ? `<figcaption>${escapeHtml(props.caption)}</figcaption>`
    : '';
  return `<figure><img src="${src}" alt="${alt}" />${caption}</figure>`;
}

function renderDivider(props: any): string {
  const style = props.style || 'solid';
  const color = props.color || 'currentColor';
  const thickness = props.thickness || 1;
  const spacing = props.spacing ?? 16;
  return `<hr style="border-style:${style};border-color:${color};border-top-width:${thickness}px;margin:${spacing}px 0;" />`;
}

function renderSpacer(props: any): string {
  const height = props.height || 32;
  return `<div style="height:${height}px;"></div>`;
}

function renderSection(props: any): string {
  const children = Array.isArray(props.content)
    ? props.content.map(renderNode).join('\n')
    : '';
  return `<section>${children}</section>`;
}

import DOMPurify from 'isomorphic-dompurify';

export function sanitizeHtml(html: string): string {
  if (!html) return '';
  return DOMPurify.sanitize(html, {
    USE_PROFILES: { html: true, svg: true, svgFilters: true },
    ADD_ATTR: ['target', 'rel', 'loading', 'class', 'style', 'data-*', 'aria-*'],
    ADD_TAGS: ['iframe'],
    FORBID_TAGS: ['script', 'object', 'embed', 'form'],
    FORBID_ATTR: ['onerror', 'onclick', 'onload', 'onmouseover', 'onfocus', 'onchange'],
  });
}

export function sanitizeCss(css: string): string {
  if (!css) return '';
  return css
    .replace(/@import[^;]+;/gi, '')
    .replace(/url\(\s*['"]?\s*javascript:[^)]*\)/gi, '')
    .replace(/expression\s*\(/gi, '')
    .replace(/behavior\s*:/gi, '');
}

function slugify(input: string): string {
  return input
    .trim()
    .toLowerCase()
    .replace(/\s+/g, '-')
    .replace(/[^a-z0-9-]/g, '')
    .replace(/-+/g, '-');
}

function serializeContent(content: PageContent): string {
  return JSON.stringify({
    html: content.html || '',
    css: content.css || '',
  });
}

/**
 * تبدیل رشته content ذخیره‌شده در DB به PageContent
 * - اگر Puck قدیمی است → مهاجرت به HTML/CSS
 * - اگر JSON { html, css } است → همان
 * - اگر HTML خام است → html = raw، css = ''
 */
function parseContent(raw: string): PageContent {
  if (!raw) return { html: '', css: '' };

  // ① Puck قدیمی → مهاجرت
  if (isLegacyPuckContent(raw)) {
    console.warn('[pageService] Legacy Puck content detected → migrating');
    return migratePuckToHtml(raw);
  }

  // ② تلاش برای JSON { html, css }
  try {
    const parsed = JSON.parse(raw);
    if (typeof parsed === 'object' && parsed !== null) {
      return {
        html: typeof parsed.html === 'string' ? parsed.html : '',
        css: typeof parsed.css === 'string' ? parsed.css : '',
      };
    }
  } catch {
    // ③ HTML خام
    return { html: raw, css: '' };
  }

  return { html: '', css: '' };
}

export const pageService = {
  /**
   * دریافت لیست صفحات
   */
  async getList(params?: {
    title?: string;
    pageNumber?: number;
    pageSize?: number;
  }): Promise<PageListItem[]> {
    try {
      const response = await apiClient.get<{
        currentPage: number;
        totalPages: number;
        pageSize: number;
        totalCount: number;
        items: Array<{
          id: string;
          title: string;
          englishTitle: string;
          url: string;
        }>;
      }>(API.STATIC_PAGES, {
        params: {
          Title: params?.title || '',
          PageNumber: params?.pageNumber || 1,
          PageSize: params?.pageSize || 100,
        },
      });

      return (response.data.items || []).map((item) => ({
        id: item.id,
        slug: item.englishTitle,
        title: item.title,
        status: 'published' as const,
        updatedAt: new Date().toISOString(),
      }));
    } catch (error) {
      console.error('[pageService] getList failed:', error);
      return [];
    }
  },

  /**
   * دریافت یک صفحه با ID
   */
  async getById(id: string): Promise<FullPage | null> {
    try {
      const response = await apiClient.get<PageApiResponse>(API.STATIC_PAGE, {
        params: { Id: id },
      });
      const data = response.data;
      if (!data || !data.id) return null;

      return {
        meta: {
          id: data.id,
          slug: data.englishTitle,
          title: data.title,
          englishTitle: data.englishTitle,
          url: data.url,
          seo: data.seoInformation
            ? {
                metaTitle: data.seoInformation.title || null,
                metaDescription: data.seoInformation.description || null,
                metaKeywords: data.seoInformation.keywords || null,
                canonicalUrl: data.seoInformation.canonicalUrl || null,
                ogImage: null,
              }
            : null,
          status: 'published',
          createdAt: data.createdAt || new Date().toISOString(),
          updatedAt: data.updatedAt || new Date().toISOString(),
        },
        content: parseContent(data.content),
      };
    } catch (error) {
      console.error('[pageService] getById failed:', error);
      return null;
    }
  },

  /**
   * دریافت یک صفحه با slug (englishTitle)
   */
  async getBySlug(slug: string): Promise<FullPage | null> {
    try {
      const list = await this.getList({ pageSize: 500 });
      const found = list.find((p) => p.slug === slug);
      if (!found) return null;
      return this.getById(found.id);
    } catch {
      return null;
    }
  },

  /**
   * ساخت صفحه جدید
   */
  async create(
    request: CreatePageRequest,
    categoryId: string
  ): Promise<FullPage | null> {
    try {
      const slug = slugify(request.meta.slug);

      const body = {
        title: request.meta.title,
        englishTitle: slug,
        url: `/page/${slug}`,
        content: serializeContent(request.content),
        categoryId,
        seoInformation: request.meta.seo
          ? {
              title: request.meta.seo.metaTitle || request.meta.title,
              description:
                request.meta.seo.metaDescription || request.meta.title,
              canonicalUrl:
                request.meta.seo.canonicalUrl || `/page/${slug}`,
            }
          : {
              title: request.meta.title,
              description: request.meta.title,
              canonicalUrl: `/page/${slug}`,
            },
      };

      const response = await apiClient.post<PageApiResponse>(
        API.STATIC_PAGE,
        body
      );

      if (!response.data?.id) return null;
      return this.getById(response.data.id);
    } catch (error) {
      console.error('[pageService] create failed:', error);
      throw error;
    }
  },

  /**
   * ویرایش صفحه
   */
  async update(request: UpdatePageRequest): Promise<FullPage | null> {
    try {
      const current = await this.getById(request.id);
      if (!current) throw new Error('صفحه یافت نشد');

      const meta = request.meta || {};
      const content = request.content || current.content;
      const slug = meta.slug ? slugify(meta.slug) : current.meta.slug;
      const title = meta.title || current.meta.title;

      const body = {
        id: request.id,
        title,
        englishTitle: slug,
        url: `/page/${slug}`,
        content: serializeContent(content),
        seoInformation: meta.seo
          ? {
              title: meta.seo.metaTitle || title,
              description: meta.seo.metaDescription || title,
              canonicalUrl: meta.seo.canonicalUrl || `/page/${slug}`,
            }
          : undefined,
        isActive: true,
      };

      await apiClient.put(API.STATIC_PAGE, body);

      return this.getById(request.id);
    } catch (error) {
      console.error('[pageService] update failed:', error);
      throw error;
    }
  },

  /**
   * حذف صفحه
   */
  async remove(id: string): Promise<boolean> {
    try {
      await apiClient.delete(API.STATIC_PAGE, { data: { id } });
      return true;
    } catch (error) {
      console.error('[pageService] remove failed:', error);
      return false;
    }
  },

  /**

        items: Array<{ id: string; title: string; isActive: boolean }>;
      }>(API.STATIC_PAGE_CATEGORIES, {
        params: { PageNumber: 1, PageSize: 100 },
      });

      return (response.data.items || []).map((c) => ({
        id: c.id,
        title: c.title,
        isActive: c.isActive,
      }));
    } catch (error) {
      console.error('[pageService] getCategories failed:', error);
      return [];
    }
  },
};
